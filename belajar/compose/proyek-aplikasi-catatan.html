Proyek: aplikasi catatan

Buka versi interaktif

Proyek akhir menggabungkan modul sebelumnya: Room menyimpan catatan, repository membungkus DAO, ViewModel mengubah Flow menjadi StateFlow, dan layar Compose menampilkan daftar dengan pencarian serta tombol tambah.

Pencarian memakai flatMapLatest: setiap kata kunci berubah, query lama dibatalkan dan diganti query baru. debounce(300) menunggu pengguna berhenti mengetik agar database tidak ditanya terlalu sering.

Kerjakan tantangan di bawah untuk melengkapinya menjadi aplikasi yang siap dipakai.

KODE
class CatatanViewModel(private val dao: CatatanDao) : ViewModel() {
    val kata = MutableStateFlow("")

    @OptIn(FlowPreview::class, ExperimentalCoroutinesApi::class)
    val daftar: StateFlow<List<Catatan>> = kata
        .debounce(300)
        .flatMapLatest { k -> if (k.isBlank()) dao.semua() else dao.cari(k) }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    fun tambah(judul: String, isi: String) = viewModelScope.launch {
        dao.tambah(Catatan(judul = judul, isi = isi))
    }

    fun hapus(c: Catatan) = viewModelScope.launch { dao.hapus(c) }
}

@Composable
fun LayarCatatan(vm: CatatanViewModel, onTambah: () -> Unit) {
    val daftar by vm.daftar.collectAsStateWithLifecycle()
    val kata by vm.kata.collectAsStateWithLifecycle()
    Scaffold(
        topBar = { TopAppBar(title = { Text("Catatan (${daftar.size})") }) },
        floatingActionButton = {
            FloatingActionButton(onClick = onTambah) { Icon(Icons.Default.Add, "Tambah catatan") }
        },
    ) { pad ->
        Column(Modifier.padding(pad).padding(horizontal = 16.dp)) {
            OutlinedTextField(
                value = kata,
                onValueChange = { vm.kata.value = it },
                label = { Text("Cari judul") },
                singleLine = true,
                modifier = Modifier.fillMaxWidth(),
            )
            if (daftar.isEmpty()) {
                Text("Belum ada catatan. Tekan + untuk menulis.", Modifier.padding(top = 24.dp))
            }
            LazyColumn(verticalArrangement = Arrangement.spacedBy(8.dp), contentPadding = PaddingValues(vertical = 12.dp)) {
                items(daftar, key = { it.id }) { c ->
                    ElevatedCard(Modifier.fillMaxWidth()) {
                        Row(Modifier.padding(16.dp), verticalAlignment = Alignment.CenterVertically) {
                            Column(Modifier.weight(1f)) {
                                Text(c.judul, style = MaterialTheme.typography.titleMedium)
                                Text(c.isi, maxLines = 2, style = MaterialTheme.typography.bodyMedium)
                            }
                            IconButton(onClick = { vm.hapus(c) }) { Icon(Icons.Default.Delete, "Hapus ${c.judul}") }
                        }
                    }
                }
            }
        }
    }
}

Soal latihan

  1. Buat layar tambah catatan dengan dua field dan tombol Simpan yang nonaktif jika judul kosong, lalu kembali ke daftar.

    Petunjuk

    rememberSaveable untuk field; enabled = judul.isNotBlank(); panggil vm.tambah lalu onSelesai().

    Lihat jawaban
    KODE
    @Composable
fun LayarTambah(vm: CatatanViewModel, onSelesai: () -> Unit) {
    var judul by rememberSaveable { mutableStateOf("") }
    var isi by rememberSaveable { mutableStateOf("") }
    Column(Modifier.padding(16.dp), verticalArrangement = Arrangement.spacedBy(12.dp)) {
        OutlinedTextField(judul, { judul = it }, label = { Text("Judul") }, modifier = Modifier.fillMaxWidth())
        OutlinedTextField(isi, { isi = it }, label = { Text("Isi") }, minLines = 5, modifier = Modifier.fillMaxWidth())
        Button(
            onClick = { vm.tambah(judul.trim(), isi.trim()); onSelesai() },
            enabled = judul.isNotBlank(),
            modifier = Modifier.fillMaxWidth(),
        ) { Text("Simpan") }
    }
}

  2. Hubungkan kedua layar dengan Navigation Compose.

    Petunjuk

    NavHost dengan rute "daftar" dan "tambah".

    Lihat jawaban
    KODE
    NavHost(nav, startDestination = "daftar") {
    composable("daftar") { LayarCatatan(vm, onTambah = { nav.navigate("tambah") }) }
    composable("tambah") { LayarTambah(vm, onSelesai = { nav.popBackStack() }) }
}

  3. Tambahkan konfirmasi sebelum menghapus memakai AlertDialog.

    Petunjuk

    Simpan catatan yang akan dihapus di state; tampilkan dialog jika tidak null.

    Lihat jawaban
    KODE
    var akanDihapus by remember { mutableStateOf<Catatan?>(null) }
akanDihapus?.let { c ->
    AlertDialog(
        onDismissRequest = { akanDihapus = null },
        title = { Text("Hapus catatan?") },
        text = { Text("\"${c.judul}\" akan dihapus permanen.") },
        confirmButton = { TextButton(onClick = { vm.hapus(c); akanDihapus = null }) { Text("Hapus") } },
        dismissButton = { TextButton(onClick = { akanDihapus = null }) { Text("Batal") } },
    )
}
// IconButton hapus: onClick = { akanDihapus = c }

Poin penting

  • Room + Flow + StateFlow + Compose.
  • debounce + flatMapLatest untuk pencarian.
  • Konfirmasi aksi yang tidak bisa dibatalkan.
