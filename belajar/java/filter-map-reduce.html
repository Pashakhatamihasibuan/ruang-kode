JAVA
List<Integer> a = List.of(1, 2, 3, 4);
List<Integer> g = a.stream().filter(n -> n % 2 == 0).toList();
List<Integer> k = g.stream().map(n -> n * n).toList();
int s = k.stream().reduce(0, Integer::sum);
