JAVA
Scanner f = new Scanner(new File("nilai.txt"));
while (f.hasNextLine()) {
    Scanner t = new Scanner(f.nextLine());
    String nama = t.next();
    int total = t.nextInt() + t.nextInt();
    System.out.println(nama + " " + total);
}
