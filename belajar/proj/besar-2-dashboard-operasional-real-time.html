Besar 2: Dashboard operasional real-time

Buka versi interaktif

Masalah. Manajer gudang ingin melihat pesanan masuk, stok kritis, dan kinerja kurir secara langsung tanpa memuat ulang halaman.

Fitur wajib:
1. Kartu KPI (pesanan hari ini, omzet, stok kritis) yang diperbarui otomatis
2. Grafik pesanan per jam dan tabel yang bisa diurutkan dan difilter
3. Data baru masuk lewat Server-Sent Events tanpa reload
4. Filter tanggal tersimpan di URL agar bisa dibagikan
5. Tampilan bisa dipakai di tablet dan dengan keyboard

Penyelesaian langkah demi langkah:
1. Buat kontrak data (Ringkasan, Pesanan) dan API tiruan dengan MSW agar frontend bisa dikembangkan tanpa backend.
2. Ambil ringkasan dengan TanStack Query; kunci query memuat rentang tanggal dari URL.
3. Buka EventSource sekali di hook useStreamPesanan; setiap event memperbarui cache dengan setQueryData sehingga semua komponen ikut berubah.
4. Tutup koneksi saat komponen dilepas; tampilkan indikator "terputus, mencoba lagi".
5. Grafik dan tabel diberi alternatif teks (ringkasan angka) untuk pembaca layar.

Standar industri yang dinilai: data real-time memperbarui satu sumber (cache query), koneksi putus ditangani, filter di URL, grafik punya alternatif teks

Arsitektur dan stack (React + TypeScript (Vite), TanStack Query, TanStack Table, Recharts, SSE, MSW untuk data tiruan, Vitest)
BagianIsi
Data awaluseQuery GET /api/ringkasan?dari=..&sampai=..
PembaruanEventSource /api/stream: setiap event "pesanan" memperbarui cache query

KODE
// src/hooks/useStreamPesanan.ts
import { useEffect, useState } from 'react';
import { useQueryClient } from '@tanstack/react-query';
import type { Ringkasan, Pesanan } from '../types';

export function useStreamPesanan(kunci: unknown[]) {
  const qc = useQueryClient();
  const [status, setStatus] = useState<'terhubung' | 'terputus'>('terhubung');
  useEffect(() => {
    const es = new EventSource('/api/stream');
    es.addEventListener('pesanan', e => {
      const p: Pesanan = JSON.parse((e as MessageEvent).data);
      qc.setQueryData<Ringkasan>(kunci, lama => lama && {
        ...lama,
        pesananHariIni: lama.pesananHariIni + 1,
        omzet: lama.omzet + p.total,
        perJam: lama.perJam.map(j => (j.jam === new Date(p.dibuat).getHours() ? { ...j, jumlah: j.jumlah + 1 } : j)),
        terbaru: [p, ...lama.terbaru].slice(0, 50),
      });
    });
    es.onopen = () => setStatus('terhubung');
    es.onerror = () => setStatus('terputus');   // EventSource otomatis mencoba lagi
    return () => es.close();
  }, [qc, JSON.stringify(kunci)]);
  return status;
}

// src/pages/Dashboard.tsx
import { useSearchParams } from 'react-router-dom';
import { useQuery } from '@tanstack/react-query';
import { BarChart, Bar, XAxis, YAxis, Tooltip, ResponsiveContainer } from 'recharts';

export default function Dashboard() {
  const [sp, setSp] = useSearchParams();
  const tgl = sp.get('tgl') ?? new Date().toISOString().slice(0, 10);
  const kunci = ['ringkasan', tgl];
  const q = useQuery({ queryKey: kunci, queryFn: () => fetch(`/api/ringkasan?tgl=${tgl}`).then(r => r.json() as Promise<Ringkasan>) });
  const status = useStreamPesanan(kunci);
  if (q.isPending) return <p aria-busy="true">Memuat dashboard…</p>;
  if (q.isError) return <p role="alert">Gagal memuat. <button onClick={() => q.refetch()}>Coba lagi</button></p>;
  const d = q.data;
  return (
    <main>
      <input type="date" value={tgl} onChange={e => setSp({ tgl: e.target.value })} aria-label="Tanggal" />
      <span role="status">{status === 'terputus' ? 'Koneksi terputus, mencoba lagi…' : 'Langsung'}</span>
      <section className="kpi">
        <article><h2>Pesanan</h2><p>{d.pesananHariIni}</p></article>
        <article><h2>Omzet</h2><p>Rp{d.omzet.toLocaleString('id-ID')}</p></article>
        <article><h2>Stok kritis</h2><p>{d.stokKritis}</p></article>
      </section>
      <figure>
        <ResponsiveContainer width="100%" height={240}>
          <BarChart data={d.perJam}><XAxis dataKey="jam" /><YAxis allowDecimals={false} /><Tooltip /><Bar dataKey="jumlah" /></BarChart>
        </ResponsiveContainer>
        <figcaption>Pesanan per jam; jam tersibuk pukul {d.perJam.reduce((a, b) => (b.jumlah > a.jumlah ? b : a)).jam}.00</figcaption>
      </figure>
    </main>
  );
}
