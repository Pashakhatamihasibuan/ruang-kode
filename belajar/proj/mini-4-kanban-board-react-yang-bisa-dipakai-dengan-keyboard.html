Mini 4: Kanban board React yang bisa dipakai dengan keyboard

Buka versi interaktif

Masalah. Tim kecil ingin papan tugas sederhana tanpa berlangganan aplikasi. Fitur seret-lepas harus punya alternatif tombol agar aksesibel (WCAG 2.2: Dragging Movements).

Fitur wajib:
1. Tiga kolom: Akan, Sedang, Selesai
2. Tambah kartu, pindah kolom dengan tombol ← → dan dengan seret
3. Data tetap ada setelah halaman dimuat ulang
4. Keadaan kosong per kolom

Penyelesaian langkah demi langkah:
1. Simpan state sebagai array kartu {id, judul, kolom}; kolom adalah turunan (filter) dari state, bukan state terpisah.
2. Buat fungsi murni pindah(kartu, arah) yang mudah diuji.
3. Sinkronkan ke localStorage dengan useEffect; baca nilai awal dengan inisialisasi malas useState(() => ...).
4. Seret memakai HTML5 drag events; tombol ← → tetap ada untuk keyboard dan layar sentuh.

Kembangkan sendiri: urutkan kartu di dalam kolom, filter per label, dan test Vitest untuk fungsi pindah.

KODE
import { useEffect, useState } from 'react';

type Kolom = 'akan' | 'sedang' | 'selesai';
type Kartu = { id: string; judul: string; kolom: Kolom };
const KOLOM: Kolom[] = ['akan', 'sedang', 'selesai'];
const LABEL = { akan: 'Akan', sedang: 'Sedang', selesai: 'Selesai' };

export function pindah(k: Kartu, arah: -1 | 1): Kartu {
  const i = Math.min(2, Math.max(0, KOLOM.indexOf(k.kolom) + arah));
  return { ...k, kolom: KOLOM[i] };
}

export default function Kanban() {
  const [kartu, setKartu] = useState<Kartu[]>(() => JSON.parse(localStorage.getItem('kanban') ?? '[]'));
  const [judul, setJudul] = useState('');
  useEffect(() => localStorage.setItem('kanban', JSON.stringify(kartu)), [kartu]);

  const ubah = (id: string, f: (k: Kartu) => Kartu) => setKartu(ks => ks.map(k => (k.id === id ? f(k) : k)));

  return (
    <main>
      <form onSubmit={e => { e.preventDefault(); if (!judul.trim()) return;
        setKartu([...kartu, { id: crypto.randomUUID(), judul: judul.trim(), kolom: 'akan' }]); setJudul(''); }}>
        <label htmlFor="j">Tugas baru</label>
        <input id="j" value={judul} onChange={e => setJudul(e.target.value)} />
        <button>Tambah</button>
      </form>
      <div className="papan">
        {KOLOM.map(kol => {
          const isi = kartu.filter(k => k.kolom === kol);
          return (
            <section key={kol} aria-label={LABEL[kol]}
              onDragOver={e => e.preventDefault()}
              onDrop={e => ubah(e.dataTransfer.getData('id'), k => ({ ...k, kolom: kol }))}>
              <h2>{LABEL[kol]} ({isi.length})</h2>
              {isi.length === 0 && <p className="kosong">Belum ada tugas.</p>}
              {isi.map(k => (
                <article key={k.id} draggable onDragStart={e => e.dataTransfer.setData('id', k.id)}>
                  {k.judul}
                  <button aria-label={`Pindah ${k.judul} ke kiri`} disabled={kol === 'akan'} onClick={() => ubah(k.id, x => pindah(x, -1))}>←</button>
                  <button aria-label={`Pindah ${k.judul} ke kanan`} disabled={kol === 'selesai'} onClick={() => ubah(k.id, x => pindah(x, 1))}>→</button>
                  <button aria-label={`Hapus ${k.judul}`} onClick={() => setKartu(kartu.filter(x => x.id !== k.id))}>✕</button>
                </article>
              ))}
            </section>
          );
        })}
      </div>
    </main>
  );
}
