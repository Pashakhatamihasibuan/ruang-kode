Mini 2: Kalkulator patungan (split bill) dengan pajak dan servis

Buka versi interaktif

Masalah. Saat makan bersama, teman-teman sering bingung membagi tagihan yang ada pajak 11% dan servis 5%, apalagi kalau pesanannya berbeda-beda.

Fitur wajib:
1. Tambah orang dan pesanannya
2. Pajak dan servis dibagi proporsional sesuai pesanan
3. Pembulatan ke Rp100 terdekat tanpa membuat total meleset
4. Validasi: angka kosong atau negatif ditolak dengan pesan jelas

Penyelesaian langkah demi langkah:
1. Simpan data sebagai array objek {nama, subtotal}; tampilan selalu dibuat ulang dari data (satu sumber kebenaran).
2. Hitung porsi tiap orang = subtotalnya ÷ total subtotal, lalu kalikan dengan total akhir.
3. Bulatkan semua orang kecuali yang terakhir; orang terakhir membayar sisanya agar jumlahnya tepat sama dengan tagihan.
4. Gunakan Intl.NumberFormat untuk Rupiah dan tampilkan error di bawah input dengan aria-live.

Kembangkan sendiri: bagikan hasil lewat tautan (simpan data di URLSearchParams) dan tombol salin ke WhatsApp.

HTML
<form id="form" novalidate>
  <label>Nama <input id="nama" required></label>
  <label>Subtotal pesanan (Rp) <input id="sub" inputmode="numeric" required></label>
  <button>Tambah</button>
  <p id="err" role="alert" aria-live="assertive"></p>
</form>
<ul id="daftar"></ul>
<p id="ringkas"></p>

<script>
const PAJAK = 0.11, SERVIS = 0.05;
const rp = new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', maximumFractionDigits: 0 });
const orang = [];

function bagi(orang) {
  const sub = orang.reduce((s, o) => s + o.subtotal, 0);
  const total = Math.round(sub * (1 + PAJAK + SERVIS));
  let terpakai = 0;
  const hasil = orang.map((o, i) => {
    if (i === orang.length - 1) return { ...o, bayar: total - terpakai };   // sisa untuk orang terakhir
    const bayar = Math.round((o.subtotal / sub) * total / 100) * 100;
    terpakai += bayar;
    return { ...o, bayar };
  });
  return { sub, total, hasil };
}

function render() {
  const { sub, total, hasil } = bagi(orang);
  document.getElementById('daftar').innerHTML = hasil
    .map(o => `<li>${o.nama}: pesan ${rp.format(o.subtotal)} → bayar <b>${rp.format(o.bayar)}</b></li>`).join('');
  document.getElementById('ringkas').textContent = orang.length
    ? `Subtotal ${rp.format(sub)} + pajak 11% + servis 5% = ${rp.format(total)}` : 'Belum ada pesanan.';
}

document.getElementById('form').addEventListener('submit', e => {
  e.preventDefault();
  const nama = document.getElementById('nama').value.trim();
  const subtotal = Number(document.getElementById('sub').value.replace(/\D/g, ''));
  const err = document.getElementById('err');
  if (!nama) return (err.textContent = 'Nama wajib diisi.');
  if (!subtotal || subtotal <= 0) return (err.textContent = 'Subtotal harus angka lebih dari 0.');
  err.textContent = '';
  orang.push({ nama, subtotal });
  e.target.reset();
  render();
});
render();
</script>
