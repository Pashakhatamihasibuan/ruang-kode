Mini 6: Pencarian instan dengan debounce dan sorot hasil

Buka versi interaktif

Masalah. Katalog 5.000 produk terasa lambat jika setiap ketikan langsung menyaring semua data dan menggambar ulang daftar.

Fitur wajib:
1. Hasil muncul saat mengetik tanpa terasa patah
2. Pencarian tidak peka huruf besar dan aksen
3. Kata yang cocok disorot dengan <mark>
4. Navigasi hasil dengan panah atas/bawah dan Enter

Penyelesaian langkah demi langkah:
1. Debounce 200 ms agar penyaringan berjalan setelah pengguna berhenti sejenak.
2. Normalisasi teks sekali di awal (toLowerCase + normalize("NFD") buang tanda aksen) dan simpan sebagai indeks.
3. Bangun HTML sorotan dengan aman: escape teks dulu, baru bungkus bagian yang cocok.
4. Gunakan pola combobox ARIA: aria-activedescendant menunjuk hasil yang aktif.

Kembangkan sendiri: pencarian fuzzy (salah ketik satu huruf tetap ketemu) dan virtualisasi daftar untuk 50.000 item.

TYPESCRIPT
export function debounce<T extends (...a: any[]) => void>(fn: T, ms = 200) {
  let t: ReturnType<typeof setTimeout>;
  return (...a: Parameters<T>) => { clearTimeout(t); t = setTimeout(() => fn(...a), ms); };
}

const norm = (s: string) => s.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
const esc = (s: string) => s.replace(/[&<>"]/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]!));

export function sorot(teks: string, kata: string) {
  const i = norm(teks).indexOf(norm(kata));
  if (!kata || i < 0) return esc(teks);
  return esc(teks.slice(0, i)) + '<mark>' + esc(teks.slice(i, i + kata.length)) + '</mark>' + esc(teks.slice(i + kata.length));
}

type Produk = { id: number; nama: string };
export function pasangPencarian(input: HTMLInputElement, list: HTMLUListElement, data: Produk[]) {
  const indeks = data.map(p => ({ ...p, kunci: norm(p.nama) }));   // normalisasi sekali saja
  let aktif = -1, hasil: Produk[] = [];

  const gambar = (kata: string) => {
    list.innerHTML = hasil.length
      ? hasil.map((p, i) => `<li role="option" id="h${p.id}" aria-selected="${i === aktif}">${sorot(p.nama, kata)}</li>`).join('')
      : `<li role="option" aria-disabled="true">Tidak ada produk "${esc(kata)}"</li>`;
    input.setAttribute('aria-activedescendant', aktif >= 0 ? `h${hasil[aktif].id}` : '');
  };
  const saring = debounce((kata: string) => {
    const k = norm(kata.trim());
    hasil = k ? indeks.filter(p => p.kunci.includes(k)).slice(0, 20) : [];
    aktif = -1; gambar(kata.trim());
  });

  input.setAttribute('role', 'combobox');
  input.addEventListener('input', () => saring(input.value));
  input.addEventListener('keydown', e => {
    if (e.key === 'ArrowDown') aktif = Math.min(hasil.length - 1, aktif + 1);
    else if (e.key === 'ArrowUp') aktif = Math.max(0, aktif - 1);
    else if (e.key === 'Enter' && aktif >= 0) { location.hash = `produk-${hasil[aktif].id}`; return; }
    else return;
    e.preventDefault(); gambar(input.value.trim());
  });
}
KELUARAN
sorot('Kopi Gayo Arabika', 'gayo')  ->  Kopi <mark>Gayo</mark> Arabika
sorot('Café Latte', 'cafe')        ->  <mark>Café</mark> Latte
sorot('<b>Teh</b>', 'teh')         ->  &lt;b&gt;<mark>Teh</mark>&lt;/b&gt;   (aman dari XSS)
