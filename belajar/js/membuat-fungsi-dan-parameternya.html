Membuat fungsi dan parameternya

Buka versi interaktif

Fungsi adalah blok kode bernama yang bisa dipanggil berulang. Tiga cara utama: function declaration (function f() {}, di-hoist), function expression (const f = function () {}), dan arrow function. Di JS fungsi adalah first-class citizen: bisa disimpan di variabel, dikirim sebagai argumen (callback), dan dikembalikan dari fungsi lain (higher-order function).

Parameter adalah nama di definisi; argumen adalah nilai saat memanggil. Argumen yang tidak dikirim bernilai undefined, kelebihan argumen diabaikan. Default parameter (function f(x = 10)) dipakai hanya saat argumen undefined (bukan null). Rest parameter (...angka) mengumpulkan sisa argumen menjadi array asli dan harus di posisi terakhir.

Primitif dikirim by value (salinan), object dikirim by sharing (referensinya disalin), sehingga fungsi bisa memutasi object milik pemanggil. Praktik baik: fungsi murni (pure) yang tidak mengubah input dan selalu memberi hasil sama untuk input sama lebih mudah diuji. Jika parameter lebih dari tiga, gunakan satu object parameter dengan destructuring.

JAVASCRIPT
function luasPersegi(sisi) { return sisi * sisi; }          // declaration
const kelilingPersegi = function (sisi) { return 4 * sisi; }; // expression
const kubus = sisi => sisi ** 3;                               // arrow
console.log(luasPersegi(4), kelilingPersegi(4), kubus(4));

function sapa(nama = 'Tamu', salam = `Halo`) { return `${salam}, ${nama}!`; }
console.log(sapa(), sapa('Rani'), sapa(undefined, 'Hai'), sapa(null));

function jumlahkan(label, ...angka) {
  return `${label}: ${angka.reduce((a, b) => a + b, 0)} dari ${angka.length} angka`;
}
console.log(jumlahkan('total', 5, 10, 15));
console.log(jumlahkan('kosong'));

function naikkanHarga(produk, persen) { produk.harga *= 1 + persen / 100; }   // memutasi input
const p = { harga: 10000 };
naikkanHarga(p, 10);
console.log(p.harga);
const naikMurni = (produk, persen) => ({ ...produk, harga: Math.round(produk.harga * (1 + persen / 100)) });
console.log(naikMurni(p, 10), p);

const terapkan = (fn, nilai) => fn(nilai);                     // higher-order
console.log(terapkan(kubus, 3), [1, 2, 3].map(luasPersegi));
console.log(sapa.length, jumlahkan.length, sapa.name);

Poin penting

  • Fungsi adalah first-class: bisa disimpan, dikirim, dikembalikan.
  • Default parameter hanya untuk undefined.
  • Rest parameter = array sisa argumen, harus terakhir.
  • Object dikirim by sharing: hindari mutasi input.
